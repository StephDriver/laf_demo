{%- comment -%} Finds the files stored alongside a document (images, PDFs, map areas). Usage: {% include folder.html doc=page %} Sets: folder_path - the document's folder in content/, e.g. _current/projects/river-path/ folder_url - the folder's web address, e.g. /projects/river-path/ folder_files - the files in that folder and its subfolders {%- endcomment -%} {%- assign folder_path = include.doc.relative_path | split: "/" | pop | join: "/" | append: "/" -%} {%- assign folder_url = include.doc.url | split: "/" | pop | join: "/" | append: "/" -%} {%- assign _collection = site.collections | where: "label", include.doc.collection | first -%} {%- assign folder_files = _collection.files | where_exp: "f", "f.path contains folder_path" -%}
